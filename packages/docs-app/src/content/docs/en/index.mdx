## Quickstart

```bash
pnpm add @chamn/engine @chamn/model @chamn/render react react-dom
```

```tsx
import { Engine, InnerComponentMeta, plugins } from '@chamn/engine';
import renderAsURL from '@chamn/render/dist/index.umd.js?url';

export function Editor() {
  return (
    <Engine plugins={plugins.DEFAULT_PLUGIN_LIST} material={InnerComponentMeta} schema={{}} renderJSUrl={renderAsURL} />
  );
}
```
